# agent-new Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `bin/agent-new` — owner-side automation of the new-project and repo-wiring runbooks (spec: `docs/specs/2026-10-15-agent-new-design.md`).

**Architecture:** One bash script with three subcommands (`project`, `protect`, `verify`); `project` composes the other two. Every step is check-then-act (re-run = resume). Tested via `tests/launcher/test-agent-new.sh` with stub `gh`/`op`/`git`/`docker`/`agent-up`/`tailscale.exe` binaries that log calls and replay canned GitHub responses.

**Tech Stack:** bash (`set -euo pipefail`), `gh` (all GitHub API work, `--jq` for extraction — no local `jq` dependency), `op`, `git`, `docker`.

## Global Constraints

- Output prefix `agent-new:`; helpers `say`/`skip`/`warn`/`die` (spec §7). `die` exits 1, usage exits 2.
- No token ever written to disk; secrets live in one command's environment, cleared after (spec §2).
- Never modify an existing ruleset — report drift and refuse (spec §5.3).
- Never invoke `tailscale funnel`; serve status must show `(tailnet only)` (spec §2).
- `agent-new` never merges, approves, or closes PRs; never pushes `main` after the ruleset exists (spec §2).
- Machine user: `--machine-user` flag > `MACHINE_USER=` in `$CFG_ROOT/agent-up.conf` > die (spec §3).
- Test harness style: `tests/launcher/test-agent-up.sh` (`check`/`refuses` helpers, temp `$HOME`, stubs first on `PATH`, `annotate` for CI).
- Commits end with `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`.

```bash
#!/usr/bin/env bash
# agent-new: owner-side automation for new projects and repo wiring
# (docs/specs/2026-10-15-agent-new-design.md). Counterpart of bin/agent-up: agent-up runs the
# sandbox with a project's read-only token; agent-new runs AS THE OWNER (gh repo admin, op
# session) and automates docs/runbooks/new-project.md + machine-user-and-rulesets.md §2-4, §6.
set -euo pipefail

HERE="$(cd "$(dirname "$(readlink -f "$0")")" && pwd)"
CFG_ROOT="${AGENT_CONTAINERS_CONFIG:-$HOME/.config/agent-containers}"

say(){ printf 'agent-new: %s\n' "$*"; }
skip(){ printf 'agent-new: SKIP (done): %s\n' "$*"; }
warn(){ printf 'agent-new: WARNING: %s\n' "$*" >&2; }
die(){ printf 'agent-new: %s\n' "$*" >&2; exit 1; }
usage(){
  cat >&2 <<'USAGE'
usage:
  agent-new project <folder> [--image base|gis|bio] [--public] [--no-research]
                             [--t3 publish|1|0] [--token-from <vault>] [--machine-user <user>]
  agent-new protect <owner>/<repo> [--machine-user <user>] [--vault <op-vault>] [--require <check>]...
  agent-new verify  <project> | <owner>/<repo> [--vault <op-vault>]
Runs AS THE OWNER. The runbooks in docs/runbooks/ remain the reference procedure.
USAGE
  exit 2
}
conf_get(){ sed -n "s/^$2=//p" "$1" | tail -n 1; }
normalise(){ printf '%s' "$1" | tr 'A-Z' 'a-z' | sed 's/[^a-z0-9_-]/-/g; s/^[^a-z0-9]*//'; }

MACHINE_USER_FLAG="" VAULT="" IMAGE="" VISIBILITY=private RESEARCH=1 T3MODE=publish TOKEN_FROM=""
REQUIRE_CHECKS=(gitleaks)

machine_user(){
  [ -n "$MACHINE_USER_FLAG" ] && { printf '%s' "$MACHINE_USER_FLAG"; return; }
  local mu=""; [ -f "$CFG_ROOT/agent-up.conf" ] && mu="$(conf_get "$CFG_ROOT/agent-up.conf" MACHINE_USER)"
  [ -n "$mu" ] || die "no machine user known: set MACHINE_USER= in $CFG_ROOT/agent-up.conf or pass --machine-user"
  printf '%s' "$mu"
}
preflight_owner(){ # prints the owner login; dies if gh is absent/anonymous/the machine user
  local who; who="$(gh api user --jq .login 2>/dev/null)" || die "gh is not logged in as you (gh auth login)"
  [ -n "$who" ] || die "gh is not logged in as you (gh auth login)"
  [ "$who" != "$(machine_user)" ] || die "gh is logged in as the machine user ($who): agent-new runs as the OWNER"
  printf '%s' "$who"
}

cmd_project(){ die "project: not implemented yet"; }
cmd_protect(){ die "protect: not implemented yet"; }
cmd_verify(){ die "verify: not implemented yet"; }

[ $# -ge 1 ] || usage
sub="$1"; shift
args=()
while [ $# -gt 0 ]; do case "$1" in
  --machine-user) MACHINE_USER_FLAG="${2:?}"; shift 2 ;;
  --vault)        VAULT="${2:?}"; shift 2 ;;
  --image)        IMAGE="${2:?}"; shift 2 ;;
  --public)       VISIBILITY=public; shift ;;
  --no-research)  RESEARCH=0; shift ;;
  --t3)           T3MODE="${2:?}"; shift 2 ;;
  --token-from)   TOKEN_FROM="${2:?}"; shift 2 ;;
  --require)      REQUIRE_CHECKS+=("${2:?}"); shift 2 ;;
  -*)             usage ;;
  *)              args+=("$1"); shift ;;
esac; done
set -- "${args[@]:-}"

case "$sub" in
  project) [ $# -eq 1 ] && [ -n "$1" ] || usage; cmd_project "$1" ;;
  protect) [ $# -eq 1 ] || usage; [[ "$1" == */* ]] || die "protect wants <owner>/<repo>"
           preflight_owner >/dev/null; cmd_protect "$1" ;;
  verify)  [ $# -eq 1 ] || usage; cmd_verify "$1" ;;
  *) usage ;;
esac
```

```bash
mu_token(){ # machine-user PAT from the project vault; owner op session; ONE call's worth
  unset OP_SERVICE_ACCOUNT_TOKEN
  op read "op://$VAULT/github-agent/token"
}
protect_collaborator(){ # $1 owner/repo
  local repo="$1" mu; mu="$(machine_user)"
  if gh api "repos/$repo/collaborators/$mu" >/dev/null 2>&1; then
    skip "$mu is a collaborator on $repo"; return
  fi
  say "inviting $mu to $repo (machine-user-and-rulesets.md step 2.1)"
  gh api -X PUT "repos/$repo/collaborators/$mu" >/dev/null
  if [ -n "$VAULT" ]; then
    say "accepting the invitation as $mu (token read from op://$VAULT/github-agent/token for this step only)"
    local tok inv
    tok="$(mu_token)"
    inv="$(GH_TOKEN="$tok" gh api /user/repository_invitations \
           --jq ".[] | select(.repository.full_name==\"$repo\") | .id" | head -n1)"
    [ -n "$inv" ] || die "no invitation for $repo visible to $mu (invite pending elsewhere?)"
    GH_TOKEN="$tok" gh api -X PATCH "/user/repository_invitations/$inv" >/dev/null
    tok=""
  else
    warn "no --vault given: accept the invitation as $mu yourself (https://github.com/$repo/invitations) or re-run with --vault agent-<project>"
  fi
}
cmd_protect(){
  protect_collaborator "$1"
}
```

```bash
NORM_JQ='[.enforcement,
 (.bypass_actors | map("\(.actor_type):\(.actor_id):\(.bypass_mode)") | sort | join(",")),
 (.rules | map(.type) | sort | join(",")),
 ((.rules[] | select(.type=="pull_request") | .parameters |
   "pr:\(.required_approving_review_count):\(.dismiss_stale_reviews_on_push):\(.require_code_owner_review):\(.require_last_push_approval)") // "pr:-"),
 ((.rules[] | select(.type=="required_status_checks") | .parameters.required_status_checks | map(.context) | sort | join("+")) // "")
] | join("|")'

checks_sorted(){ printf '%s\n' "${REQUIRE_CHECKS[@]}" | sort -u; }
ruleset_json(){
  local checks="" c
  while IFS= read -r c; do checks+="{\"context\": \"$c\"},"; done < <(checks_sorted)
  cat <<JSON
{
  "name": "protect-main", "target": "branch", "enforcement": "active",
  "bypass_actors": [{"actor_id": 5, "actor_type": "RepositoryRole", "bypass_mode": "pull_request"}],
  "conditions": {"ref_name": {"include": ["~DEFAULT_BRANCH"], "exclude": []}},
  "rules": [
    {"type": "deletion"},
    {"type": "non_fast_forward"},
    {"type": "pull_request", "parameters": {
      "required_approving_review_count": 1, "dismiss_stale_reviews_on_push": true,
      "require_code_owner_review": true, "require_last_push_approval": true,
      "required_review_thread_resolution": false,
      "allowed_merge_methods": ["merge", "squash", "rebase"]}},
    {"type": "required_status_checks", "parameters": {
      "strict_required_status_checks_policy": false,
      "required_status_checks": [${checks%,}]}}
  ]
}
JSON
}
ruleset_norm_expected(){
  printf 'active|RepositoryRole:5:pull_request|deletion,non_fast_forward,pull_request,required_status_checks|pr:1:true:true:true|%s' \
    "$(checks_sorted | paste -sd+ -)"
}
protect_ruleset(){ # $1 owner/repo
  local repo="$1" id have
  id="$(gh api "repos/$repo/rulesets" --jq '.[] | select(.name=="protect-main") | .id' 2>/dev/null | head -n1)"
  if [ -z "$id" ]; then
    say "creating ruleset protect-main on $repo (machine-user-and-rulesets.md step 4)"
    ruleset_json | gh api -X POST "repos/$repo/rulesets" --input - >/dev/null
    return
  fi
  have="$(gh api "repos/$repo/rulesets/$id" --jq "$NORM_JQ")"
  if [ "$have" = "$(ruleset_norm_expected)" ]; then
    skip "ruleset protect-main on $repo matches the template"
  else
    warn "live:     $have"
    warn "template: $(ruleset_norm_expected)"
    die "ruleset protect-main on $repo differs from the template: refusing to modify it (never weaken). Edit it in the GitHub UI if the difference is intended."
  fi
}
protect_warnings(){ # $1 owner/repo
  local repo="$1"
  gh api "repos/$repo/contents/.github/CODEOWNERS" >/dev/null 2>&1 \
    || warn "$repo has no .github/CODEOWNERS: code-owner review cannot bind (copy project-template/repo/.github/, runbook step 2.2)"
  gh api "repos/$repo/contents/.github/workflows/gitleaks.yml" >/dev/null 2>&1 \
    || warn "$repo has no gitleaks workflow: the required check will never report (copy project-template/repo/.github/)"
  [ "$(gh api "repos/$repo/actions/permissions/workflow" --jq .can_approve_pull_request_reviews 2>/dev/null)" = true ] \
    && warn "$repo lets Actions approve PRs: turn it OFF (Settings -> Actions -> General, runbook step 2.3)" || true
}
cmd_protect(){
  protect_collaborator "$1"
  protect_ruleset "$1"
  protect_warnings "$1"
  say "protect: done ($1)"
}
```
